#rom debug
#save after-first-dungeon
# DX HP regen (estimated, include/dx_tuning.h): walking 20 turns heals 2% of max HP per turn (Red: 0.5%)
mash START sMainMenu 4 != 0 3000
wait 60
press UP
press A
wait 30
# Debug Mode -> Warp -> Tiny Woods -> 1F
press A
wait 30
press A
wait 30
press A
mash A gDungeon 4 != 0 1800
wait 600
expect sMainLoopsUnk 4 == 1
read maxhp ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoMaxHP 2
# the wind counter ticks once per turn: use it as a turn clock
write *gDungeon+@gDxOffWindTurns 2 500
# 999 max HP so wild hits can't end the run (and the HP never tops out): DX heals 2%/turn, Red 0.5%
write ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoMaxHP 2 999
write ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoHP 2 50
read hp0 ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoHP 2
read wind0 *gDungeon+@gDxOffWindTurns 2
# hold A+B: pass turns in place (no walking, so the floor layout and wild Pokemon do not matter)
hold A+B 900
read wind *gDungeon+@gDxOffWindTurns 2
read hp ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoHP 2
shot 1_after_walk
# HP gained vs turns taken: DX heals ~20 a turn (2% of 999), Red 5. At least 3 a turn, allowing for wild hits.
calc gain hp - hp0
calc turns wind0 - wind
calc want turns * 3
cmpreads gain >= want
calc minturns 0 + 10
cmpreads turns >= minturns
