# DX Rare Qualities Leave Half (eating a Big Apple leaves an Apple, an Apple a Tiny Apple), Forge a Path (the leader

# cures another)
mash START sMainMenu 4 != 0 3000
write gDxTestSeed 4 99991
# an empty toolbox, then a Big Apple (83) in slot 0
write *gTeamInventoryRef+0 1 0
write *gTeamInventoryRef+4 1 0
write *gTeamInventoryRef+8 1 0
write *gTeamInventoryRef+12 1 0
write *gTeamInventoryRef+16 1 0
write *gTeamInventoryRef+20 1 0
write *gTeamInventoryRef+24 1 0
write *gTeamInventoryRef+28 1 0
write *gTeamInventoryRef+32 1 0
write *gTeamInventoryRef+36 1 0
write *gTeamInventoryRef+40 1 0
write *gTeamInventoryRef+44 1 0
write *gTeamInventoryRef+48 1 0
write *gTeamInventoryRef+52 1 0
write *gTeamInventoryRef+56 1 0
write *gTeamInventoryRef+60 1 0
write *gTeamInventoryRef+64 1 0
write *gTeamInventoryRef+68 1 0
write *gTeamInventoryRef+72 1 0
write *gTeamInventoryRef+76 1 0

press A

press A
wait 30
press A

press A

shot 1_in_dungeon

write *gRecruitedPokemonRef+3733 1 11
press B

press DOWN
press A

press A

shot 2_item_menu
# Eat
press A

shot 4_after

expect *gTeamInventoryRef+0 1 != 0
expect *gTeamInventoryRef+2 1 == 82
# now Forge a Path (7): walk up into the wall above the room
write *gRecruitedPokemonRef+3733 1 7
read belly0 ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoBelly 2
read y0 **gDungeon+@gDxOffTeamPokemon+@gDxOffEntityPos!2 2
hold UP 400
wait 60
shot 5_tunnel
read belly1 ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoBelly 2
read y1 **gDungeon+@gDxOffTeamPokemon+@gDxOffEntityPos!2 2
# it went 10+ tiles straight up through the wall, and paid much more Belly than walking does (5 per wall)
calc climbed y0 - y1
calc paid belly0 - belly1
calc ten 10 + 0
cmpreads climbed >= ten
cmpreads paid >= ten
# every wall broken was redrawn as the leader stepped in
read digs gDxTestCount+10 2
read redrawn gDxTestCount+12 2
expect gDxTestCount+10 2 >= 2
cmpreads redrawn == digs
# Shared Happiness (25): leader and partner poisoned; the leader eats a Pecha Berry (66) and both are cured
write *gRecruitedPokemonRef+3733 1 25
write *gTeamInventoryRef+2 1 66
write ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoBurn 1 2
write ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoBurn!1 1 20
write ***gDungeon+@gDxOffTeamPokemon!4+@gDxOffEntityInfo+@gDxOffInfoBurn 1 2
write ***gDungeon+@gDxOffTeamPokemon!4+@gDxOffEntityInfo+@gDxOffInfoBurn!1 1 20
press B
wait 30
press DOWN
press A
wait 40
press A
wait 40
shot 6_pecha_menu
press A
wait 200
shot 7_cured
expect ***gDungeon+@gDxOffTeamPokemon+@gDxOffEntityInfo+@gDxOffInfoBurn 1 == 0
expect ***gDungeon+@gDxOffTeamPokemon!4+@gDxOffEntityInfo+@gDxOffInfoBurn 1 == 0
# Leave Half (11) again, on an Apple (82): a Tiny Apple (the old Beat-Up Orb slot, 236) is left
write *gRecruitedPokemonRef+3733 1 11

wait 30
press DOWN
press A
wait 40
press A
wait 40

wait 200
press A
wait 60
shot 9_tiny_left

# and the Tiny Apple fills 25 Belly, leaving nothing

wait 30
press DOWN
press A
wait 40
press A
wait 40

press A
