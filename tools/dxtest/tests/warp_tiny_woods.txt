#rom debug
#save after-first-dungeon
# Debug Mode -> Warp -> #0 Tiny Woods -> floor 1, then check we are in the dungeon
# Dungeon offsets come from the debug ROM's gDxOff* symbols
mash START sMainMenu 4 != 0 3000
wait 60
press UP
press A
wait 30
expect *sMainMenu+4 4 == 16
press A
wait 30
expect *sMainMenu+4 4 == 47
shot 1_pick_dungeon
press A
wait 30
shot 2_pick_floor
press A
mash A gDungeon 4 != 0 1800
wait 300
shot 3_in_dungeon
expect *gDungeon+@gDxOffDungeonLocation 1 == 0
expect *gDungeon+@gDxOffDungeonFloor 1 == 1
expect *gDungeon+@gDxOffFloorCount 4 == 4
